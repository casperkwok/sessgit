"""统计目录下各类文件的代码行数，按行数从多到少输出。

import os
import sys
from collections import defaultdict

EXCLUDE_DIRS = {".git", "node_modules"}
# 无扩展名时按文件名归类
SPECIAL_NAMES = {
    "Dockerfile": "Dockerfile",
    "Makefile": "Makefile",
    "Makefile.am": "Makefile",
    "CMakeLists.txt": "CMake",
    "BUILD": "BUILD",
    "Rakefile": "Ruby",
    "Gemfile": "Ruby",
}


def ext_of(filename):
    """返回文件的类型标识，如 .py / .js；无扩展名返回文件名本身。"""
    base = os.path.basename(filename)
    if base in SPECIAL_NAMES:
        return SPECIAL_NAMES[base]
    _, ext = os.path.splitext(base)
    return ext or base


def count_lines(path):
    """统计单个文件的非空行数；二进制文件返回 None。"""
    try:
        with open(path, "r", encoding="utf-8", errors="strict") as f:
            return sum(1 for line in f if line.strip())
    except (UnicodeDecodeError, OSError):
        return None


def scan(root):
    """遍历 root，返回 {类型: [文件数, 代码行数]}。"""
    stats = defaultdict(lambda: [0, 0])
    for dirpath, dirnames, filenames in os.walk(root):
        dirnames[:] = [d for d in dirnames if d not in EXCLUDE_DIRS]
        for name in filenames:
            n = count_lines(os.path.join(dirpath, name))
            if n is None:  # 二进制或不可读，跳过
                continue
            stat = stats[ext_of(name)]
            stat[0] += 1
            stat[1] += n
    return stats


def main():
    root = sys.argv[1] if len(sys.argv) > 1 else "."
    if not os.path.isdir(root):
        sys.exit(f"错误: 不是目录: {root}")

    stats = scan(root)
    if not stats:
        print(f"{root} 下没有可统计的文本文件")
        return

    rows = sorted(stats.items(), key=lambda kv: kv[1][1], reverse=True)
    width = max(len(k) for k, _ in rows)
    print(f"{'类型'.ljust(width)}  {'文件数':>6}  {'代码行数':>8}")
    print("-" * (width + 18))
    for ext, (files, lines) in rows:
        print(f"{ext.ljust(width)}  {files:>6}  {lines:>8}")
    print("-" * (width + 18))
    total_files = sum(v[0] for v in stats.values())
    total_lines = sum(v[1] for v in stats.values())
    print(f"{'合计'.ljust(width)}  {total_files:>6}  {total_lines:>8}")


if __name__ == "__main__":
    main()

def skip_string(text, i, quote):
    """i 指向起始引号，返回字符串结束后的下标。"""
    i += 1
    while i < len(text):
        if text[i] == "\\":
            i += 2
        elif text[i] == quote:
            return i + 1
        else:
            i += 1
    return i


def analyze_line(text, syntax, block_end):
    """剔除注释，返回 (本行是否还有代码, 跨行块注释的新状态)。

    只按标记切分，不解析语法：字符串字面量会整体跳过（所以 `"http://x"`
    里的 `//` 不会误判成注释），但正则字面量、heredoc 之类仍可能误判。
    """
    line_tokens, block_pairs = syntax
    i, n = 0, len(text)
    has_code = False
    while i < n:
        if block_end is not None:  # 已在块注释里，只找结束标记
            if text.startswith(block_end, i):
                i += len(block_end)
                block_end = None
            else:
                i += 1
            continue

        for start, end in block_pairs:
            if text.startswith(start, i):
                # Python 三引号只在行首出现时算 docstring；赋值语句里的
                # 三引号是字符串，按字符串处理。

                block_end = end
                i += len(start)
                break

                i = skip_string(text, i, text[i])
                continue

# countloc

统计目录下各类文件的代码行数，按行数从多到少输出。单文件，只用标准库。

## 用法

```
python3 countloc.py [目录] [--top N] [--exclude-comments]
```

| 参数 | 说明 |
| --- | --- |
| `目录` | 要统计的目录，默认当前目录 |
| `--top N` | 只显示行数最多的前 N 类；被截掉的几类合并成一行「其余 N 类」显示，合计仍是全量 |
| `--exclude-comments` | 剔除注释后再统计，只对认识的注释语法生效 |

```console
$ python3 countloc.py --top 3
类型       文件数  代码行数
---------------------------
.py             1       197
.js             1         7
Makefile        1         3
其余 2 类       2         5
---------------------------
合计            4       212
```

## 统计口径

- **代码行** = 去掉空行后的行数。加 `--exclude-comments` 后，整行都是注释的行（含块注释内的行、Python docstring）也不计入；行尾注释不影响该行。
- **首行 shebang 算代码行**，例如 `#!/bin/bash` 在 `--exclude-comments` 下仍计入 `.sh`。
- **排除** `.git` 和 `node_modules` 目录；二进制文件（UTF-8 解码失败）跳过。
- **分组**按扩展名；无扩展名时按文件名（`Makefile`、`Dockerfile`、`CMakeLists.txt` 等单独成组）。文件名为 `BUILD` 也单独成组。
- **排序**按代码行数降序，行数相同时顺序不保证稳定。

## 两个取舍

**一、注释识别是标记切分，不是真解析器。** 逐字符扫，只认标记，不建语法树：

- 认识的语法才剔：`#`（Python / Shell / YAML / Makefile 等）、`//` `/* */`（C 系 / JS / Go / Rust 等）、`--` `--[[ ]]`（SQL / Lua / Haskell）、`;`（Lisp 系）、`<!-- -->`（HTML / XML / Vue）。
- 不认识的类型（`.md`、`.json`、`.csv`、`.txt`）原样统计，不做任何剔除。
- 字符串字面量会整体跳过，所以 `"http://x/a#b"` 里的 `//` 和 `#` 不会误判成注释。
- 但 JS 正则字面量（`/\/\//`）、heredoc、未加引号的 CSS `url(http://…)` 仍可能误判 —— 这类行通常是既有代码又有"注释"标记，代码部分已经置位，所以影响一般落在"少算"而不是"多算"。
- Python 三引号只在行首（前面还没有代码）时按 docstring 剔除；`x = """…"""` 赋值里的三引号按字符串算，是启发式规则，不是语法层判断。

**二、只统计文本文件，二进制直接跳过。** 判断方式是"能不能用 UTF-8 严格解码"，所以 GBK 编码的源码会被当成二进制跳过（换编码得改 `open()` 的 `encoding`）。这样做是为了不引入 charset 探测依赖，代价是编码不是 UTF-8 的文件会静默消失，不进任何一列。
